# Example /etc/config/videoplayer after install

config videoplayer 'main'
	option enabled '1'
	# USB stick mounted at /mnt/sda1:
	option media_path '/mnt/sda1/videos'
	option allow_remote '1'
	option render_mode 'browser'
	# Fast favors throughput; Quality preserves the 640x360 renderer path.
	option router_profile 'fast'
	# Router CPU renderer FPS: 5, 8, 12, 15, 20, 24, 30, 48, 50, or 60
	option router_fps '8'
	# WARNING: 1 uses maximum FFmpeg-supported CPU parallelism and can make the
	# router unstable or interrupt internet access while rendering.
	option router_max_threads '0'
	option max_depth '8'
